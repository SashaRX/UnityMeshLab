cmake_minimum_required(VERSION 3.20)
project(meshlab-intel-query-bench LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(EMBREE_ROOT "" CACHE PATH "Pinned Embree CPU binary distribution")
if(NOT WIN32 OR NOT EXISTS "${EMBREE_ROOT}/include/embree4/rtcore.h")
    message(FATAL_ERROR "Supply a Windows Embree 4 CPU distribution")
endif()
add_library(meshlab-embree-bench SHARED embree_query_bench.cpp)
target_include_directories(meshlab-embree-bench PRIVATE "${EMBREE_ROOT}/include")
target_link_libraries(meshlab-embree-bench PRIVATE "${EMBREE_ROOT}/lib/embree4.lib")
target_compile_options(meshlab-embree-bench PRIVATE /O2 /fp:strict)
foreach(runtime embree4 tbb12 tbbmalloc)
    add_custom_command(TARGET meshlab-embree-bench POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${EMBREE_ROOT}/bin/${runtime}.dll" "$<TARGET_FILE_DIR:meshlab-embree-bench>")
endforeach()

set(IPP_DEVEL_ROOT "" CACHE PATH "Private Intel IPP NuGet development extraction")
set(IPP_REDIST_ROOT "" CACHE PATH "Private Intel IPP NuGet runtime extraction")
if(IPP_DEVEL_ROOT AND IPP_REDIST_ROOT)
    add_executable(meshlab-ipp-mask-bench ipp_mask_bench.cpp)
    target_include_directories(meshlab-ipp-mask-bench PRIVATE "${IPP_DEVEL_ROOT}/build/native/include/ipp")
    target_link_libraries(meshlab-ipp-mask-bench PRIVATE "${IPP_DEVEL_ROOT}/build/native/win-x64/ippcv.lib"
        "${IPP_DEVEL_ROOT}/build/native/win-x64/ippcore.lib")
    target_compile_options(meshlab-ipp-mask-bench PRIVATE /O2)
    file(GLOB IPP_RUNTIMES "${IPP_REDIST_ROOT}/runtimes/win-x64/native/ippcv*.dll" "${IPP_REDIST_ROOT}/runtimes/win-x64/native/ippcore.dll")
    foreach(runtime IN LISTS IPP_RUNTIMES)
        add_custom_command(TARGET meshlab-ipp-mask-bench POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
            "${runtime}" "$<TARGET_FILE_DIR:meshlab-ipp-mask-bench>")
    endforeach()
endif()
